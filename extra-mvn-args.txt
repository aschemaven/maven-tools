# Per-project extra Maven CLI arguments appended to the mvn invocation.
# Format: <project> <arg1> [<arg2> ...]
# Comments starting with `#` and blank lines are ignored.
#
# Use sparingly -- this is the catch-all hatch for one-off Surefire,
# Failsafe, or plugin-specific flags that we don't want global. Prefer
# the dedicated per-project lists (isolated-m2-projects.txt,
# lrm-split-skip-projects.txt, develocity-skip-projects.txt,
# flaky-projects.txt) for their respective concerns.

# wagon-providers tests occasionally hang in surefire's forked JVM
# (HttpsWagon TLS handshake / HTTP keep-alive corner cases on macOS
# loopback). Cap each surefire fork at 10 minutes so a hang turns into
# a test failure (which the retry mechanism can then ride over) rather
# than an infinite stall that requires manual kill -9.
misc/wagon -Dsurefire.timeout=600

# Maven core IT suites (its/core-it-suite) spawn sub-Maven processes
# from each IT. Sub-mvns regularly hang (Develocity extension issues,
# OOM in inner forks, Jetty test servers not shutting down, ...) and
# Surefire has no fork timeout configured. Result: the outer mvn waits
# forever for a fork that will never complete. Cap surefire forks at
# 10 minutes so a stuck IT becomes a measurable failure instead of an
# overnight stall (observed: core/maven-4.0.x hung 13h after a
# MavenITmng6759 OOM in a sub-fork during RC-6 prep).
core/maven-4.0.x -Dsurefire.timeout=600
core/3.x/its-3 -Dsurefire.timeout=600
core/3.x/mvnd-1 -Dsurefire.timeout=600
core/mvnd -Dsurefire.timeout=600

# E2ETest is a throughput benchmark, not a correctness test: it asserts that a
# writer thread completes 400k encode calls inside a 30s latch. Upstream calls
# it flaky itself (b2d4904bd, 2026-03-12, "relax timeout which makes test
# really flaky") and tripled a sibling limit rather than fix it. Measured here:
# it FAILED on 4 of 5 columns at ~36s during the loaded 2026-10-26 sweep and
# PASSED on all 5 at ~13s on a quiet box. It sits in module 12 of 17, so it
# stops the reactor before maven-surefire-its -- the only module in this
# component that measures Maven compatibility at all.
plugins/core/surefire -Dsurefire.excludes=**/E2ETest.java
